72void ImageInstanceSegmentation::setupDiagnostics() {
73 declareAndLoadParameter("diagnostic_updater.use_fqn", diagnostic_use_fqn_,
74 "Prefix diagnostic status names with the fully qualified node name");
75 declareAndLoadParameter("diagnostic_updater.period", diagnostic_period_s_, "Diagnostic reporting period in seconds");
76 declareAndLoadParameter("diagnostic_updater.inputs.timeout", input_timeout_s_,
77 "Required camera receive timeout in steady seconds");
78 declareAndLoadParameter("diagnostic_updater.inputs.max_age", input_max_age_s_, "Maximum input acquisition age in ROS seconds");
79 declareAndLoadParameter("diagnostic_updater.inputs.min_age", input_min_age_s_, "Minimum ROS now minus image stamp in seconds");
80 processing_timeout_s_ = std::max(5.0, 2.0 * triton_client_timeout_s_);
81 declareAndLoadParameter("diagnostic_updater.processing.timeout", processing_timeout_s_,
82 "Maximum time without successful processing in steady seconds");
83 declareAndLoadParameter("diagnostic_updater.processing.warn_duration", processing_warn_duration_s_,
84 "Input callback to output duration warning threshold in seconds");
85 for (const auto value :
86 {diagnostic_period_s_, input_timeout_s_, input_max_age_s_, processing_timeout_s_, processing_warn_duration_s_}) {
87 if (!std::isfinite(value) || value <= 0.0)
88 throw std::invalid_argument("Diagnostic periods and timeouts must be finite and positive");
89 }
90 if (!diagnostic_use_fqn_) throw std::invalid_argument("diagnostic_updater.use_fqn must be true");
91 if (!std::isfinite(input_min_age_s_) || input_min_age_s_ > input_max_age_s_) {
92 throw std::invalid_argument("Diagnostic input timestamp bounds must be finite and ordered");
93 }
94 if (processing_warn_duration_s_ > processing_timeout_s_) {
95 throw std::invalid_argument("Diagnostic processing warn_duration must not exceed timeout");
96 }
97 if (diagnostic_period_s_ < 0.001) throw std::invalid_argument("Diagnostic reporting period must be at least 0.001 seconds");
98 diagnostic_updater_ = std::make_unique<diagnostic_updater::Updater>(this, diagnostic_period_s_, Status::STALE);
99 diagnostic_updater_->setHardwareID("none");
100 for (auto& [name, state] : camera_health_) {
101 const auto& camera = cameras_.at(name);
102 const std::map<std::string, std::string> topics{{"input", camera.image_subscriber.getTopic()},
103 {"class_mask", camera.class_mask_publisher->get_topic_name()},
104 {"instance_mask", camera.instance_mask_publisher->get_topic_name()},
105 {"detections", camera.detections_publisher->get_topic_name()}};
106 for (const auto& [key, topic] : topics) {
107 auto& cfg = state.topics[key];
108 const std::string prefix = "diagnostic_updater." + name + "." + key + ".";
109 declareAndLoadParameter(prefix + "nominal_frequency", cfg.nominal_frequency,
110 "Expected topic frequency in Hz; zero disables the periodic topic task");
111 cfg.min_frequency = 0.8 * cfg.nominal_frequency;
112 cfg.max_frequency = 1.2 * cfg.nominal_frequency;
113 declareAndLoadParameter(prefix + "min_frequency", cfg.min_frequency, "Minimum topic frequency in Hz");
114 declareAndLoadParameter(prefix + "max_frequency", cfg.max_frequency, "Maximum topic frequency in Hz");
115 if (!std::isfinite(cfg.nominal_frequency) || !std::isfinite(cfg.min_frequency) || !std::isfinite(cfg.max_frequency) ||
116 cfg.nominal_frequency < 0.0 || cfg.min_frequency < 0.0 || cfg.max_frequency < cfg.min_frequency ||
117 (cfg.nominal_frequency > 0.0 && cfg.min_frequency <= 0.0) ||
118 (cfg.nominal_frequency == 0.0 && (cfg.min_frequency != 0.0 || cfg.max_frequency != 0.0))) {
119 throw std::invalid_argument("Invalid frequency bounds for " + prefix);
120 }
121 if (cfg.nominal_frequency > 0.0) {
122 cfg.task = std::make_unique<CapturedTopic>(
123 topic, *diagnostic_updater_, cfg.min_frequency, cfg.max_frequency, input_min_age_s_,
124 input_max_age_s_ + (key == "input" ? 0.0 : processing_timeout_s_), get_clock(), cfg.result);
125 }
126 }
127 }
128 diagnostic_updater_->add("health", this, &ImageInstanceSegmentation::health);
129}

132 auto& state = camera_health_.at(camera_name);

140void ImageInstanceSegmentation::health(Wrapper& status) {
141 struct Check {
142 unsigned char level = Status::OK;
143 std::string message = "All checks OK";
144 };
145 std::map<std::string, Check> checks{{"inputs", {}}, {"processing", {}}, {"data_integrity", {}}};
146 Wrapper details;
147 const auto steady_now = std::chrono::steady_clock::now();
148 const auto ros_now = now();
149 const auto assess = [&](const std::string& key, unsigned char level, const std::string& message) {
150 auto& check = checks[key];
151 if (severity(level) > severity(check.level)) check = {level, message};
152 };
153 for (auto& [name, state] : camera_health_) {
154 CameraHealthSnapshot snapshot;
155 {
156 std::lock_guard<std::mutex> lock(state.mutex);
157 snapshot = state;
158 state.failure_in_cycle = false;
159 state.invalid_in_cycle = false;
160 state.peak_duration_s = 0.0;
161 }
162 const double receive_age =
163 std::chrono::duration<double>(steady_now - (snapshot.received ? snapshot.last_receive : diagnostics_started_at_)).count();
164 const double output_age =
165 std::chrono::duration<double>(steady_now - (snapshot.completed
166 ? snapshot.last_success
167 : (snapshot.received ? snapshot.first_receive : diagnostics_started_at_)))
168 .count();
169 const double input_age = snapshot.received ? (ros_now - snapshot.input_stamp).seconds() : 0.0;
170 if (receive_age > input_timeout_s_) {
171 assess("inputs", Status::ERROR, name + ": required image receive timeout");
172 } else if (!snapshot.received) {
173 assess("inputs", Status::STALE, name + ": awaiting first image");
174 } else if (input_age < input_min_age_s_ || input_age > input_max_age_s_) {
175 assess("inputs", Status::ERROR, name + ": image acquisition timestamp outside bounds");
176 }
177 if (!snapshot.received) assess("data_integrity", Status::STALE, name + ": awaiting first image");
178 if (!snapshot.input_valid || snapshot.invalid_in_cycle)
179 assess("data_integrity", Status::ERROR, name + ": invalid image dimensions, data, frame or timestamp");
180 if (snapshot.failure_in_cycle || snapshot.processing_failed) {
181 assess("processing", Status::ERROR, name + ": " + snapshot.processing_error);
182 } else if (!snapshot.backend_ready) {
183 assess("processing", Status::STALE, name + ": awaiting Triton initialization");
184 } else if (snapshot.received && output_age > processing_timeout_s_) {
185 assess("processing", Status::ERROR, name + ": no successful result within processing timeout");
186 } else if (!snapshot.completed) {
187 assess("processing", Status::STALE, name + ": awaiting first result");
188 } else if (std::max(snapshot.duration_s, snapshot.peak_duration_s) > processing_warn_duration_s_) {
189 assess("processing", Status::WARN, name + ": processing duration above warning threshold");
190 }
191 if (snapshot.initialization_failed) {
192 assess("initialization", snapshot.initialization_error.empty() ? Status::OK : Status::ERROR,
193 name + ": " + snapshot.initialization_error);
194 details.add("initialization." + name + ".error", snapshot.initialization_error);
195 }
196 details.add("inputs." + name + ".received_since_start", snapshot.received);
197 details.add("inputs." + name + ".receive_age_s", receive_age);
198 if (snapshot.received) details.add("inputs." + name + ".acquisition_age_s", input_age);
199 details.add("data_integrity." + name + ".invalid_metadata_since_start", snapshot.invalid);
200 details.add("processing." + name + ".backend_ready", snapshot.backend_ready);
201 details.add("processing." + name + ".completed_since_start", snapshot.completed);
202 details.add("processing." + name + ".failed_since_start", snapshot.failed);
203 details.add("processing." + name + ".dropped_busy_since_start", snapshot.dropped);
204 details.add("processing." + name + ".dropped_not_ready_since_start", snapshot.dropped_not_ready);
205 details.add("processing." + name + ".in_flight", cameras_.at(name).in_flight.load());
206 details.add("processing." + name + ".output_age_s", output_age);
207 details.add("processing." + name + ".duration_s", snapshot.duration_s);
208 details.add("processing." + name + ".peak_duration_since_report_s", snapshot.peak_duration_s);
209 for (const auto& [key, topic] : state.topics) {
210 if (!topic.task) continue;
211 const std::string check = "topic." + name + "." + key;
212 checks[check] = {topic.result.level,
213 topic.result.message.empty() ? "Frequency and timestamps within bounds" : topic.result.message};
214 static const std::map<std::string, std::string> fields{
215 {"Events in window", "events_in_window"},
216 {"Events since startup", "events_since_start"},
217 {"Duration of window (s)", "window_s"},
218 {"Actual frequency (Hz)", "frequency_hz"},
219 {"Target frequency (Hz)", "target_frequency_hz"},
220 {"Minimum acceptable frequency (Hz)", "min_frequency_hz"},
221 {"Maximum acceptable frequency (Hz)", "max_frequency_hz"},
222 {"Earliest timestamp delay:", "min_delay_s"},
223 {"Latest timestamp delay:", "max_delay_s"},
224 {"Earliest acceptable timestamp delay:", "min_allowed_delay_s"},
225 {"Latest acceptable timestamp delay:", "max_allowed_delay_s"},
226 {"Late diagnostic update count:", "late_reports_since_start"},
227 {"Early diagnostic update count:", "early_reports_since_start"},
228 {"Zero seen diagnostic update count:", "zero_stamp_reports_since_start"}};
229 for (const auto& value : topic.result.values) {
230 const auto field = fields.find(value.key);
231 if (field != fields.end()) details.add(check + "." + field->second, value.value);
232 }
233 }
234 }
235 Check worst;
236 for (const auto& [key, check] : checks) {
237 status.add(key, std::string(levelName(check.level)) + ": " + check.message);
238 if (severity(check.level) > severity(worst.level)) worst = {check.level, key + ": " + check.message};
239 }
240 status.summary(worst.level, worst.message);
241 status.values.insert(status.values.end(), details.values.begin(), details.values.end());
242}
