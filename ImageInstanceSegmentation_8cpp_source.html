78// Fixed-size pool of persistent worker threads. Replaces spawning a new
79// std::thread per frame, which under sustained camera frame rates repeatedly
80// pays thread create/teardown cost for no benefit. Each camera has its own
81// Triton client and mutex, so cameras run concurrently here; only same-camera
82// in-flight requests (up to max_in_flight_per_camera_) serialize against each
83// other.
84class ThreadPool {
85 public:
88 explicit ThreadPool(std::size_t worker_count) {
89 worker_count = std::max<std::size_t>(1, worker_count);
90 workers_.reserve(worker_count);
91 for (std::size_t i = 0; i < worker_count; ++i) {
92 workers_.emplace_back([this] { workerLoop(); });
93 }
94 }
95
97 ThreadPool(const ThreadPool&) = delete;
99 ThreadPool& operator=(const ThreadPool&) = delete;
100
103 ~ThreadPool() {
104 {
105 std::lock_guard<std::mutex> lock(mutex_);
106 stopping_ = true;
107 }
108 condition_.notify_all();
109 for (auto& worker : workers_) {
110 if (worker.joinable()) {
111 worker.join();
112 }
113 }
114 }
115
117 void enqueue(std::function<void()> task) {
118 {
119 std::lock_guard<std::mutex> lock(mutex_);
120 tasks_.push_back(std::move(task));
121 }
122 condition_.notify_one();
123 }
124
125 private:
127 void workerLoop() {
128 while (true) {
129 std::function<void()> task;
130 {
131 std::unique_lock<std::mutex> lock(mutex_);
132 condition_.wait(lock, [this] { return stopping_ || !tasks_.empty(); });
133 if (tasks_.empty()) {
134 if (stopping_) {
135 return;
136 }
137 continue;
138 }
139 task = std::move(tasks_.front());
140 tasks_.pop_front();
141 }
142 task();
143 }
144 }
145
146 std::vector<std::thread> workers_;
147 std::deque<std::function<void()>> tasks_;
148 std::mutex mutex_;
149 std::condition_variable condition_;
150 bool stopping_ = false;
151};

288cv::Mat rawImageToBgr(const sensor_msgs::msg::Image& msg) {
289 namespace enc = sensor_msgs::image_encodings;
290
291 if (msg.width == 0 || msg.height == 0 || msg.width > static_cast<uint32_t>(std::numeric_limits<int>::max()) ||
292 msg.height > static_cast<uint32_t>(std::numeric_limits<int>::max())) {
293 throw std::runtime_error("image dimensions must be positive and fit into OpenCV dimensions");
294 }
295
296 auto checkStep = [&](std::size_t minimum_step) {
297 if (msg.step < minimum_step) {
298 throw std::runtime_error("image step is smaller than width * pixel size");
299 }
300 if (msg.data.size() < static_cast<std::size_t>(msg.step) * msg.height) {
301 throw std::runtime_error("image data buffer is smaller than step * height");
302 }
303 };
304
305 if (msg.encoding == enc::BGR8 || msg.encoding == "8UC3") {
306 checkStep(static_cast<std::size_t>(msg.width) * 3);
307 return cv::Mat(static_cast<int>(msg.height), static_cast<int>(msg.width), CV_8UC3, const_cast<std::uint8_t*>(msg.data.data()),
308 msg.step);
309 }
310 if (msg.encoding == enc::RGB8) {
311 checkStep(static_cast<std::size_t>(msg.width) * 3);
312 cv::Mat rgb(static_cast<int>(msg.height), static_cast<int>(msg.width), CV_8UC3, const_cast<std::uint8_t*>(msg.data.data()),
313 msg.step);
314 cv::Mat bgr;
315 cv::cvtColor(rgb, bgr, cv::COLOR_RGB2BGR);
316 return bgr;
317 }
318 if (msg.encoding == enc::BGRA8 || msg.encoding == "8UC4") {
319 checkStep(static_cast<std::size_t>(msg.width) * 4);
320 cv::Mat bgra(static_cast<int>(msg.height), static_cast<int>(msg.width), CV_8UC4, const_cast<std::uint8_t*>(msg.data.data()),
321 msg.step);
322 cv::Mat bgr;
323 cv::cvtColor(bgra, bgr, cv::COLOR_BGRA2BGR);
324 return bgr;
325 }
326 if (msg.encoding == enc::RGBA8) {
327 checkStep(static_cast<std::size_t>(msg.width) * 4);
328 cv::Mat rgba(static_cast<int>(msg.height), static_cast<int>(msg.width), CV_8UC4, const_cast<std::uint8_t*>(msg.data.data()),
329 msg.step);
330 cv::Mat bgr;
331 cv::cvtColor(rgba, bgr, cv::COLOR_RGBA2BGR);
332 return bgr;
333 }
334 if (msg.encoding == enc::MONO8 || msg.encoding == "8UC1") {
335 checkStep(static_cast<std::size_t>(msg.width));
336 cv::Mat mono(static_cast<int>(msg.height), static_cast<int>(msg.width), CV_8UC1, const_cast<std::uint8_t*>(msg.data.data()),
337 msg.step);
338 cv::Mat bgr;
339 cv::cvtColor(mono, bgr, cv::COLOR_GRAY2BGR);
340 return bgr;
341 }
342 if (msg.encoding == enc::MONO16 || msg.encoding == "16UC1") {
343 checkStep(static_cast<std::size_t>(msg.width) * 2);
344 cv::Mat mono16(static_cast<int>(msg.height), static_cast<int>(msg.width), CV_16UC1,
345 const_cast<std::uint8_t*>(msg.data.data()), msg.step);
346 const uint16_t endian_marker = 1;
347 const bool host_big_endian = *reinterpret_cast<const uint8_t*>(&endian_marker) == 0;
348 if (static_cast<bool>(msg.is_bigendian) != host_big_endian) {
349 cv::Mat native(mono16.size(), CV_16UC1);
350 for (int y = 0; y < native.rows; ++y) {
351 const auto* source = msg.data.data() + static_cast<std::size_t>(y) * msg.step;
352 auto* destination = native.ptr<uint16_t>(y);
353 for (int x = 0; x < native.cols; ++x) {
354 destination[x] = msg.is_bigendian ? (static_cast<uint16_t>(source[2 * x]) << 8) | source[2 * x + 1]
355 : (static_cast<uint16_t>(source[2 * x + 1]) << 8) | source[2 * x];
356 }
357 }
358 mono16 = std::move(native);
359 }
360 cv::Mat mono8;
361 mono16.convertTo(mono8, CV_8UC1, 1.0 / 256.0);
362 cv::Mat bgr;
363 cv::cvtColor(mono8, bgr, cv::COLOR_GRAY2BGR);
364 return bgr;
365 }
366
367 throw std::runtime_error("unsupported raw image encoding '" + msg.encoding + "'");
368}

393cv::KalmanFilter makeKalmanFilter(const cv::Rect2f& box) {
394 cv::KalmanFilter filter(8, 4, 0, CV_32F);
395 filter.transitionMatrix = cv::Mat::eye(8, 8, CV_32F);
396 for (int i = 0; i < 4; ++i) {
397 filter.transitionMatrix.at<float>(i, i + 4) = 1.0f;
398 }
399 filter.measurementMatrix = cv::Mat::zeros(4, 8, CV_32F);
400 for (int i = 0; i < 4; ++i) {
401 filter.measurementMatrix.at<float>(i, i) = 1.0f;
402 }
403 cv::setIdentity(filter.processNoiseCov, cv::Scalar(1e-2));
404 cv::setIdentity(filter.measurementNoiseCov, cv::Scalar(1e-1));
405 cv::setIdentity(filter.errorCovPost, cv::Scalar(1.0));
406 filter.statePost = cv::Mat::zeros(8, 1, CV_32F);
407 const auto measurement = measurementFromBox(box);
408 for (int i = 0; i < 4; ++i) {
409 filter.statePost.at<float>(i, 0) = measurement[i];
410 }
411 filter.statePre = filter.statePost.clone();
412 return filter;
413}

417std::vector<int> solveAssignment(const std::vector<std::vector<float>>& cost_matrix) {
418 const int rows = static_cast<int>(cost_matrix.size());
419 const int cols = rows > 0 ? static_cast<int>(cost_matrix.front().size()) : 0;
420 if (rows == 0 || cols == 0) {
421 return std::vector<int>(rows, -1);
422 }
423
424 const int size = std::max(rows, cols);
425 const double kPadCost = 1e3;
426 std::vector<std::vector<double>> cost(size + 1, std::vector<double>(size + 1, kPadCost));
427 for (int i = 0; i < rows; ++i) {
428 for (int j = 0; j < cols; ++j) {
429 cost[i + 1][j + 1] = static_cast<double>(cost_matrix[static_cast<std::size_t>(i)][static_cast<std::size_t>(j)]);
430 }
431 }
432
433 std::vector<double> u(size + 1, 0.0);
434 std::vector<double> v(size + 1, 0.0);
435 std::vector<int> p(size + 1, 0);
436 std::vector<int> way(size + 1, 0);
437
438 for (int i = 1; i <= size; ++i) {
439 p[0] = i;
440 int j0 = 0;
441 std::vector<double> minv(size + 1, std::numeric_limits<double>::infinity());
442 std::vector<bool> used(size + 1, false);
443 do {
444 used[j0] = true;
445 const int i0 = p[j0];
446 double delta = std::numeric_limits<double>::infinity();
447 int j1 = 0;
448 for (int j = 1; j <= size; ++j) {
449 if (used[j]) {
450 continue;
451 }
452 const double cur = cost[i0][j] - u[i0] - v[j];
453 if (cur < minv[j]) {
454 minv[j] = cur;
455 way[j] = j0;
456 }
457 if (minv[j] < delta) {
458 delta = minv[j];
459 j1 = j;
460 }
461 }
462 for (int j = 0; j <= size; ++j) {
463 if (used[j]) {
464 u[p[j]] += delta;
465 v[j] -= delta;
466 } else {
467 minv[j] -= delta;
468 }
469 }
470 j0 = j1;
471 } while (p[j0] != 0);
472
473 do {
474 const int j1 = way[j0];
475 p[j0] = p[j1];
476 j0 = j1;
477 } while (j0 != 0);
478 }
479
480 std::vector<int> assignment(rows, -1);
481 for (int j = 1; j <= size; ++j) {
482 if (p[j] >= 1 && p[j] <= rows && j <= cols) {
483 assignment[static_cast<std::size_t>(p[j] - 1)] = j - 1;
484 }
485 }
486 return assignment;
487}

497 std::vector<Detection> update(const std::vector<Detection>& detections) override {
498 for (auto& track : tracks_) {
499 if (track.state == TrackLifecycle::Removed) {
500 continue;
501 }
502 track.predicted_box = boxFromState(track.filter.predict());
503 track.updated_this_frame = false;
504 }
505
506 std::vector<int> high_detection_indices;
507 std::vector<int> low_detection_indices;
508 high_detection_indices.reserve(detections.size());
509 low_detection_indices.reserve(detections.size());
510 for (std::size_t i = 0; i < detections.size(); ++i) {
511 const auto& detection = detections[i];
512 if (detection.box.area() < static_cast<float>(config_.min_box_area_px)) {
513 continue;
514 }
515 if (detection.score >= static_cast<float>(config_.high_confidence_threshold)) {
516 high_detection_indices.push_back(static_cast<int>(i));
517 } else if (detection.score >= static_cast<float>(config_.low_confidence_threshold)) {
518 low_detection_indices.push_back(static_cast<int>(i));
519 }
520 }
521
522 std::vector<int> candidate_track_indices;
523 candidate_track_indices.reserve(tracks_.size());
524 for (std::size_t i = 0; i < tracks_.size(); ++i) {
525 if (tracks_[i].state != TrackLifecycle::Removed) {
526 candidate_track_indices.push_back(static_cast<int>(i));
527 }
528 }
529
530 auto first_stage = associate(candidate_track_indices, high_detection_indices, detections);
531 for (const auto& match : first_stage.matches) {
532 updateTrack(tracks_[static_cast<std::size_t>(match.first)], detections[static_cast<std::size_t>(match.second)]);
533 }
534
535 std::vector<int> second_stage_tracks;
536 second_stage_tracks.reserve(first_stage.unmatched_track_indices.size());
537 for (int track_index : first_stage.unmatched_track_indices) {
538 const auto state = tracks_[static_cast<std::size_t>(track_index)].state;
539 if (state == TrackLifecycle::Tracked || state == TrackLifecycle::Lost) {
540 second_stage_tracks.push_back(track_index);
541 }
542 }
543
544 auto second_stage = associate(second_stage_tracks, low_detection_indices, detections);
545 for (const auto& match : second_stage.matches) {
546 updateTrack(tracks_[static_cast<std::size_t>(match.first)], detections[static_cast<std::size_t>(match.second)]);
547 }
548
549 for (int detection_index : first_stage.unmatched_detection_indices) {
550 const auto& detection = detections[static_cast<std::size_t>(detection_index)];
551 if (detection.score >= static_cast<float>(config_.new_track_confidence_threshold)) {
552 createTrack(detection);
553 }
554 }
555
556 std::vector<Detection> tracked_detections;
557 for (auto& track : tracks_) {
558 if (track.state == TrackLifecycle::Removed) {
559 continue;
560 }
561 if (track.updated_this_frame) {
562 if (track.state == TrackLifecycle::Tentative && track.hits >= 2) {
563 track.state = TrackLifecycle::Tracked;
564 }
565 track.latest_detection.track_id = track.id;
566 track.latest_detection.is_tracked = true;
567 tracked_detections.push_back(track.latest_detection);
568 continue;
569 }
570
571 if (track.state == TrackLifecycle::Tentative) {
572 retireTrack(track);
573 continue;
574 }
575
576 ++track.lost_frames;
577 if (track.lost_frames > config_.max_lost_frames) {
578 retireTrack(track);
579 } else {
580 track.state = TrackLifecycle::Lost;
581 }
582 }
583
584 tracks_.erase(
585 std::remove_if(tracks_.begin(), tracks_.end(), [](const Track& track) { return track.state == TrackLifecycle::Removed; }),
586 tracks_.end());
587
588 std::sort(tracked_detections.begin(), tracked_detections.end(),
589 [](const Detection& lhs, const Detection& rhs) { return lhs.track_id < rhs.track_id; });
590 return tracked_detections;
591 }

620 AssociationResult associate(const std::vector<int>& track_indices,
621 const std::vector<int>& detection_indices,
622 const std::vector<Detection>& detections) const {
623 AssociationResult result;
624 if (track_indices.empty()) {
625 result.unmatched_detection_indices = detection_indices;
626 return result;
627 }
628 if (detection_indices.empty()) {
629 result.unmatched_track_indices = track_indices;
630 return result;
631 }
632
633 const float max_cost = 1.0f - static_cast<float>(config_.match_iou_threshold);
634 const float blocked_cost = 1e3f;
635 std::vector<std::vector<float>> cost_matrix(track_indices.size(), std::vector<float>(detection_indices.size(), blocked_cost));
636
637 for (std::size_t row = 0; row < track_indices.size(); ++row) {
638 const auto& track = tracks_[static_cast<std::size_t>(track_indices[row])];
639 for (std::size_t col = 0; col < detection_indices.size(); ++col) {
640 const auto& detection = detections[static_cast<std::size_t>(detection_indices[col])];
641 if (track.latest_detection.class_index != detection.class_index) {
642 continue;
643 }
644 const float iou = intersectionOverUnion(track.predicted_box, detection.box);
645 if (iou < static_cast<float>(config_.match_iou_threshold)) {
646 continue;
647 }
648 cost_matrix[row][col] = 1.0f - iou;
649 }
650 }
651
652 const auto assignment = solveAssignment(cost_matrix);
653 std::vector<bool> detection_taken(detection_indices.size(), false);
654 for (std::size_t row = 0; row < track_indices.size(); ++row) {
655 const int assigned_col = assignment[row];
656 if (assigned_col >= 0 && assigned_col < static_cast<int>(detection_indices.size()) &&
657 cost_matrix[row][static_cast<std::size_t>(assigned_col)] <= max_cost) {
658 result.matches.emplace_back(track_indices[row], detection_indices[static_cast<std::size_t>(assigned_col)]);
659 detection_taken[static_cast<std::size_t>(assigned_col)] = true;
660 } else {
661 result.unmatched_track_indices.push_back(track_indices[row]);
662 }
663 }
664
665 for (std::size_t col = 0; col < detection_indices.size(); ++col) {
666 if (!detection_taken[col]) {
667 result.unmatched_detection_indices.push_back(detection_indices[col]);
668 }
669 }
670 return result;
671 }
672
675 void updateTrack(Track& track, const Detection& detection) {
676 cv::Mat measurement(4, 1, CV_32F);
677 const auto measurement_values = measurementFromBox(detection.box);
678 for (int i = 0; i < 4; ++i) {
679 measurement.at<float>(i, 0) = measurement_values[i];
680 }
681 track.filter.correct(measurement);
682 track.predicted_box = boxFromState(track.filter.statePost);
683 track.latest_detection = detection;
684 track.latest_detection.box = track.predicted_box;
685 track.latest_detection.track_id = track.id;
686 track.latest_detection.is_tracked = true;
687 track.updated_this_frame = true;
688 track.lost_frames = 0;
689 ++track.hits;
690 if (track.state == TrackLifecycle::Lost || track.state == TrackLifecycle::Tracked) {
691 track.state = TrackLifecycle::Tracked;
692 }
693 }

698 const std::uint16_t id = allocateTrackId();

703 Track track;

707 track.latest_detection.track_id = id;
708 track.latest_detection.is_tracked = true;

1072void ImageInstanceSegmentation::initializeModelForCamera(const std::string& camera_name) {
1073 std::lock_guard<std::mutex> lock(model_mutexes_.at(camera_name));
1074 auto& state = camera_health_.at(camera_name);
1075 {
1076 std::lock_guard<std::mutex> state_lock(state.mutex);
1077 state.backend_ready = false;
1078 }
1079 constexpr auto kRetryDelay = std::chrono::seconds(1);
1080 while (!stopping_.load() && rclcpp::ok(get_node_base_interface()->get_context())) {
1081 try {
1082 // Discover fixed output dimensions using a standard inference before
1083 // allocating buffers. The public triton_cpp API exposes metadata shapes,
1084 // which may contain -1.
1085 const auto output_shape_overrides = probeOutputShapes();
1086
1087 auto interface = std::make_unique<triton_cpp::TritonInterface>(model_name_, model_version_, server_url_, use_shm_, false,
1088 false, triton_client_timeout_s_, false);
1089 (void)interface->getInputShape(input_tensor_name_);
1090 for (const auto& name : outputTensorNames()) (void)interface->getOutputShape(name);
1091 // The model's declared input shape may have dynamic (-1) height/width
1092 // dims, which the standard (non-variable-size) input path can't size
1093 // correctly on its own. Pin it to the actual shape we always send so the
1094 // input buffer is allocated with the right size.
1095 interface->initInOutputs(output_shape_overrides, std::map<std::string, std::vector<int64_t>>{
1096 {input_tensor_name_, {1, 3, input_height_, input_width_}}});
1097 triton_interfaces_.at(camera_name) = std::move(interface);
1098 {
1099 std::lock_guard<std::mutex> state_lock(state.mutex);
1100 state.backend_ready = true;
1101 state.initialization_error.clear();
1102 }
1103 RCLCPP_INFO(get_logger(),
1104 "Camera '%s': connected to Triton model '%s:%s' at '%s' "
1105 "using input dtype '%s' and output dtype '%s'",
1106 camera_name.c_str(), model_name_.c_str(), model_version_.c_str(), server_url_.c_str(),
1107 tensorDataTypeName(input_tensor_data_type_).c_str(), tensorDataTypeName(output_tensor_data_type_).c_str());
1108 return;
1109 } catch (const std::exception& e) {
1110 {
1111 std::lock_guard<std::mutex> state_lock(state.mutex);
1112 state.initialization_failed = true;
1113 state.initialization_error = e.what();
1114 }
1115 RCLCPP_WARN(get_logger(),
1116 "Camera '%s': failed to initialize Triton model '%s:%s' at "
1117 "'%s': %s. Retrying in 1s.",
1118 camera_name.c_str(), model_name_.c_str(), model_version_.c_str(), server_url_.c_str(), e.what());
1119 // Component unloading also interrupts retries while the ROS context stays alive.
1120 const auto retry_at = std::chrono::steady_clock::now() + kRetryDelay;
1121 while (!stopping_.load() && std::chrono::steady_clock::now() < retry_at &&
1122 rclcpp::ok(get_node_base_interface()->get_context())) {
1123 std::this_thread::sleep_for(std::chrono::milliseconds(50));
1124 }
1125 }
1126 }
1127 throw std::runtime_error("Shutdown requested while waiting for Triton connection");
1128}

1167cv::Mat ImageInstanceSegmentation::preprocessImage(const cv::Mat& bgr_image, LetterboxInfo& letterbox) const {
1168 // RF-DETR is trained on images resized without preserving the aspect ratio.
1169 if (architecture_ == ModelArchitecture::RfDetr) {
1170 return cv::dnn::blobFromImage(bgr_image, 1.0 / 255.0, cv::Size(input_width_, input_height_), cv::Scalar(), true, false,
1171 CV_32F);
1172 }
1173 letterbox.scale =
1174 std::min(static_cast<double>(input_width_) / bgr_image.cols, static_cast<double>(input_height_) / bgr_image.rows);
1175 letterbox.resized_width = std::clamp(static_cast<int>(std::round(bgr_image.cols * letterbox.scale)), 1, input_width_);
1176 letterbox.resized_height = std::clamp(static_cast<int>(std::round(bgr_image.rows * letterbox.scale)), 1, input_height_);
1177 letterbox.pad_x = (input_width_ - letterbox.resized_width) / 2;
1178 letterbox.pad_y = (input_height_ - letterbox.resized_height) / 2;
1179
1180 cv::Mat letterboxed(input_height_, input_width_, CV_8UC3, cv::Scalar(114, 114, 114));
1181 cv::Mat resized = letterboxed(cv::Rect(letterbox.pad_x, letterbox.pad_y, letterbox.resized_width, letterbox.resized_height));
1182 cv::resize(bgr_image, resized, resized.size());
1183 return cv::dnn::blobFromImage(letterboxed, 1.0 / 255.0, cv::Size(), cv::Scalar(), true, false, CV_32F);
1184}

1186template <typename TensorScalar>
1189void copyInputTensor(triton_cpp::TritonInterface& triton_interface,
1190 const std::string& tensor_name,
1191 int input_height,
1192 int input_width,
1193 const cv::Mat& blob) {
1194 const std::size_t element_count =
1195 static_cast<std::size_t>(1) * 3 * static_cast<std::size_t>(input_height) * static_cast<std::size_t>(input_width);
1196 const std::size_t byte_size = element_count * sizeof(TensorScalar);
1197 const auto* input_data = blob.ptr<float>();
1198
1199 if (triton_interface.usesCudaInputSharedMemory()) {
1200 if constexpr (std::is_same_v<TensorScalar, float>) {
1201 triton_interface.copyInputTensorToDevice(tensor_name, input_data, byte_size);
1202 } else {
1203 std::vector<TensorScalar> converted(element_count);
1204 for (std::size_t i = 0; i < element_count; ++i) {
1205 converted[i] = TensorScalar(input_data[i]);
1206 }
1207 triton_interface.copyInputTensorToDevice(tensor_name, converted.data(), byte_size);
1208 }
1209 return;
1210 }
1211 if constexpr (std::is_same_v<TensorScalar, float>) {
1212 auto input = triton_interface.getInputTensor<TensorScalar>(tensor_name, 1, 3, input_height, input_width);
1213 auto& first_elem = input(0, 0, 0, 0);
1214 auto* dest_ptr = &first_elem;
1215 std::memcpy(dest_ptr, input_data, byte_size);
1216 } else {
1217 auto input = triton_interface.getInputTensor<TensorScalar>(tensor_name, 1, 3, input_height, input_width);
1218 auto& first_elem = input(0, 0, 0, 0);
1219 auto* dest_ptr = &first_elem;
1220 for (std::size_t i = 0; i < element_count; ++i) {
1221 dest_ptr[i] = TensorScalar(input_data[i]);
1222 }
1223 }
1224}

1233std::map<std::string, std::vector<int64_t>> ImageInstanceSegmentation::probeOutputShapes() {
1234 // Use the public Triton gRPC API to obtain shapes from an actual result.
1235 // The configured input dimensions and exported output dimensions must remain
1236 // fixed.
1237 namespace tc = triton::client;
1238 const auto check = [](const tc::Error& error) {
1239 if (!error.IsOk()) throw std::runtime_error(error.Message());
1240 };
1241 std::unique_ptr<tc::InferenceServerGrpcClient> client;
1242 check(tc::InferenceServerGrpcClient::Create(&client, server_url_));
1243 tc::InferOptions options(model_name_);
1244 options.model_version_ = model_version_;
1245 options.client_timeout_ = static_cast<uint64_t>(triton_client_timeout_s_ * 1e6);
1246
1247 const std::size_t count = static_cast<std::size_t>(3) * input_height_ * input_width_;
1248 const bool fp16 = input_tensor_data_type_ == TensorDataType::Fp16;
1249 const std::vector<uint8_t> zeros(count * (fp16 ? sizeof(uint16_t) : sizeof(float)), 0);
1250 tc::InferInput* raw_input = nullptr;
1251 check(tc::InferInput::Create(&raw_input, input_tensor_name_, {1, 3, input_height_, input_width_}, fp16 ? "FP16" : "FP32"));
1252 std::unique_ptr<tc::InferInput> input(raw_input);
1253 check(input->AppendRaw(zeros.data(), zeros.size()));
1254
1255 const auto output_names = outputTensorNames();
1256 std::vector<std::unique_ptr<tc::InferRequestedOutput>> outputs;
1257 std::vector<const tc::InferRequestedOutput*> raw_outputs;
1258 for (const auto& name : output_names) {
1259 tc::InferRequestedOutput* output = nullptr;
1260 check(tc::InferRequestedOutput::Create(&output, name));
1261 outputs.emplace_back(output);
1262 raw_outputs.push_back(output);
1263 }
1264 tc::InferResult* raw_result = nullptr;
1265 check(client->Infer(&raw_result, options, {input.get()}, raw_outputs));
1266 std::unique_ptr<tc::InferResult> result(raw_result);
1267 check(result->RequestStatus());
1268 std::map<std::string, std::vector<int64_t>> shapes;
1269 for (const auto& name : output_names) {
1270 check(result->Shape(name, &shapes[name]));
1271 if (shapes[name].empty() ||
1272 std::any_of(shapes[name].begin(), shapes[name].end(), [](int64_t dimension) { return dimension <= 0; })) {
1273 throw std::runtime_error("Model output must have positive, fixed dimensions: " + name);
1274 }
1275 }
1276 return shapes;
1277}

1279void ImageInstanceSegmentation::imageCallback(const sensor_msgs::msg::Image::ConstSharedPtr& msg,
1280 const std::string& camera_name) {
1281 const auto input_received_at = std::chrono::steady_clock::now();
1282 auto& state = camera_health_.at(camera_name);
1283 bool backend_ready = false;
1284 const bool valid_stamp = msg->header.stamp.sec >= 0 && msg->header.stamp.nanosec < 1000000000u;
1285 {
1286 std::lock_guard<std::mutex> state_lock(state.mutex);
1287 backend_ready = state.backend_ready;
1288 if (!backend_ready) ++state.dropped_not_ready;
1289 if (state.received == 0) state.first_receive = input_received_at;
1290 state.last_receive = input_received_at;
1291 state.input_stamp = valid_stamp ? rclcpp::Time(msg->header.stamp, get_clock()->get_clock_type())
1292 : rclcpp::Time(0, 0, get_clock()->get_clock_type());
1293 ++state.received;
1294 state.input_valid = msg->width > 0 && msg->height > 0 && !msg->data.empty() && !msg->header.frame_id.empty() && valid_stamp &&
1295 state.input_stamp.nanoseconds() > 0;
1296 if (!state.input_valid) {
1297 ++state.invalid;
1298 state.invalid_in_cycle = true;
1299 }
1300 }
1301 if (auto& task = state.topics.at("input").task; task && valid_stamp) task->tick(msg->header.stamp);
1302 // Receive health includes all frames, but initialization must not queue stale images.
1303 if (!backend_ready) return;
1304 std::lock_guard<std::mutex> admission_lock(admission_mutex_);
1305 auto& runtime = cameras_.at(camera_name);
1306 if (stopping_.load()) return;
1307 int in_flight = runtime.in_flight.load();
1308 do {
1309 if (in_flight >= max_in_flight_per_camera_) {
1310 std::lock_guard<std::mutex> state_lock(state.mutex);
1311 ++state.dropped;
1312 return;
1313 }
1314 } while (!runtime.in_flight.compare_exchange_weak(in_flight, in_flight + 1));
1315
1316 try {
1317 thread_pool_->enqueue([this, msg, camera_name, input_received_at, &runtime] {
1318 struct ReleaseSlot {
1319 std::atomic<int>& count;
1320 ~ReleaseSlot() { count.fetch_sub(1); }
1321 } release{runtime.in_flight};
1322 if (stopping_.load()) return;
1323 try {
1324 // msg owns the storage behind raw BGR views throughout processing.
1325 processImage(msg->header, rawImageToBgr(*msg), camera_name, input_received_at);
1326 } catch (const std::exception& error) {
1327 recordProcessingFailure(camera_name, error.what());
1328 RCLCPP_WARN_THROTTLE(get_logger(), *get_clock(), 5000, "Camera '%s': processing failed: %s", camera_name.c_str(),
1329 error.what());
1330 }
1331 });
1332 } catch (...) {
1333 runtime.in_flight.fetch_sub(1);
1334 throw;
1335 }
1336}

1338void ImageInstanceSegmentation::processImage(const std_msgs::msg::Header& header,
1339 const cv::Mat& bgr_image,
1340 const std::string& camera_name,
1341 std::chrono::steady_clock::time_point input_received_at) {
1342 auto& camera = cameras_.at(camera_name);
1343 // A camera's Triton buffers and tracker are shared by its queued requests.
1344 std::unique_lock<std::mutex> lock(model_mutexes_.at(camera_name));
1345 if (stopping_.load()) return;
1346 LetterboxInfo letterbox;
1347 const cv::Mat blob = preprocessImage(bgr_image, letterbox);
1348 auto& interface = *triton_interfaces_.at(camera_name);
1349 auto inference_started_at = std::chrono::steady_clock::now();
1350 try {
1351 switch (input_tensor_data_type_) {
1352 case TensorDataType::Fp32:
1353 copyInputTensor<float>(interface, input_tensor_name_, input_height_, input_width_, blob);
1354 break;
1355 case TensorDataType::Fp16:
1356 copyInputTensor<Eigen::half>(interface, input_tensor_name_, input_height_, input_width_, blob);
1357 break;
1358 }
1359 inference_started_at = std::chrono::steady_clock::now();
1360 interface.infer();
1361 } catch (const std::exception& error) {
1362 recordProcessingFailure(camera_name, error.what());
1363 lock.unlock();
1364 RCLCPP_WARN_THROTTLE(get_logger(), *get_clock(), 5000, "Camera '%s': Triton request failed: %s. Reinitializing.",
1365 camera_name.c_str(), error.what());
1366 initializeModelForCamera(camera_name);
1367 return;
1368 }
1369 const auto inference_finished_at = std::chrono::steady_clock::now();
1370 std::shared_lock<std::shared_mutex> parameters_lock(runtime_parameters_mutex_);
1371 const double score_threshold =
1372 tracker_config_.enabled ? std::min(confidence_threshold_, tracker_config_.low_confidence_threshold) : confidence_threshold_;
1373 bool invalid_result = false;
1374 auto detections = decodeDetections(interface, bgr_image.cols, bgr_image.rows, letterbox, score_threshold, invalid_result);
1375 if (invalid_result) recordProcessingFailure(camera_name, "Non-finite detection candidates in model output");
1376 if (camera.tracker) detections = camera.tracker->update(detections);
1377 const auto masks = reconstructMasks(interface, detections, bgr_image.cols, bgr_image.rows, letterbox);
1378 // Keep publication in the same critical section as inference/tracking.
1379 publishResults(camera, header, bgr_image, detections, masks);
1380 auto& state = camera_health_.at(camera_name);
1381 for (const auto* key : {"class_mask", "instance_mask", "detections"}) {
1382 if (auto& task = state.topics.at(key).task; task && header.stamp.sec >= 0 && header.stamp.nanosec < 1000000000u)
1383 task->tick(header.stamp);
1384 }
1385 {
1386 std::lock_guard<std::mutex> state_lock(state.mutex);
1387 const auto finished_at = std::chrono::steady_clock::now();
1388 state.duration_s = std::chrono::duration<double>(finished_at - input_received_at).count();
1389 state.peak_duration_s = std::max(state.peak_duration_s, state.duration_s);
1390 state.processing_failed = invalid_result;
1391 if (!invalid_result) {
1392 state.last_success = finished_at;
1393 ++state.completed;
1394 }
1395 }
1396 if (print_timing_) {
1397 const auto finished_at = std::chrono::steady_clock::now();
1398 RCLCPP_INFO(get_logger(),
1399 "Timing camera '%s': inference=%.3f ms, postprocessing=%.3f "
1400 "ms, input_to_output=%.3f ms",
1401 camera_name.c_str(),
1402 std::chrono::duration<double, std::milli>(inference_finished_at - inference_started_at).count(),
1403 std::chrono::duration<double, std::milli>(finished_at - inference_finished_at).count(),
1404 std::chrono::duration<double, std::milli>(finished_at - input_received_at).count());
1405 }

1427std::vector<Detection> ImageInstanceSegmentation::decodeYoloDetections(triton_cpp::TritonInterface& triton_interface,
1428 int image_width,
1429 int image_height,
1430 const LetterboxInfo& letterbox,
1431 double score_threshold,
1432 bool& invalid_result) {
1433 const bool yolo26 = architecture_ == ModelArchitecture::Yolo26;
1434 const auto runtime_shape = triton_interface.getOutputShape(yolo_detections_output_tensor_name_);
1435 if (!((runtime_shape.size() == 3 && runtime_shape[0] == 1) || runtime_shape.size() == 2)) {
1436 throw std::runtime_error(
1437 "YOLO detection output must have shape [1, "
1438 "channels, anchors] or [1, anchors, channels]");
1439 }
1440
1441 const std::array<int64_t, 3> shape = runtime_shape.size() == 3
1442 ? std::array<int64_t, 3>{runtime_shape[0], runtime_shape[1], runtime_shape[2]}
1443 : std::array<int64_t, 3>{1, runtime_shape[0], runtime_shape[1]};
1444 const int64_t axis1 = shape[1];
1445 const int64_t axis2 = shape[2];
1446 const int num_classes = static_cast<int>(class_names_.size());
1447 const auto proto_shape = triton_interface.getOutputShape(yolo_prototypes_output_tensor_name_);
1448 if (!((proto_shape.size() == 4 && proto_shape[0] == 1) || proto_shape.size() == 3)) {
1449 throw std::runtime_error("YOLO prototype output must have shape [1, masks, height, width]");
1450 }
1451 const int64_t mask_count = proto_shape.size() == 4 ? proto_shape[1] : proto_shape[0];
1452 // YOLO26: x1, y1, x2, y2, sigmoid confidence, class id. YOLOv8/v11: cx, cy, w, h, class scores.
1453 const int64_t mask_channel_offset = yolo26 ? 6 : 4 + num_classes;
1454 const int64_t expected_channels = mask_channel_offset + mask_count;
1455 if ((axis1 == expected_channels) == (axis2 == expected_channels)) {
1456 throw std::runtime_error(
1457 "Detection dimensions must identify exactly one "
1458 "box/class/mask channel axis");
1459 }
1460 const bool channels_first = axis1 == expected_channels;
1461 const int64_t anchors = channels_first ? axis2 : axis1;
1462
1463 std::vector<Detection> detections;
1464 auto decode_with_tensor = [&](auto tensor) {
1465 auto value_at = [&](int64_t channel, int64_t anchor) -> float {
1466 return static_cast<float>(channels_first ? tensor(0, channel, anchor) : tensor(0, anchor, channel));
1467 };
1468
1469 for (int64_t anchor = 0; anchor < anchors; ++anchor) {
1470 int best_class = 0;
1471 float best_score = -std::numeric_limits<float>::infinity();
1472 if (yolo26) {
1473 best_score = value_at(4, anchor);
1474 const float class_id = value_at(5, anchor);
1475 if (!std::isfinite(class_id) || class_id < 0 || class_id >= num_classes || std::floor(class_id) != class_id) {
1476 invalid_result = true;
1477 continue;
1478 }
1479 best_class = static_cast<int>(class_id);
1480 } else {
1481 for (int cls = 0; cls < num_classes; ++cls) {
1482 const float score = value_at(4 + cls, anchor);
1483 if (!std::isfinite(score)) invalid_result = true;
1484 if (score > best_score) {
1485 best_score = score;
1486 best_class = cls;
1487 }
1488 }
1489 }
1490 if (!std::isfinite(best_score)) invalid_result = true;
1491 if (!std::isfinite(best_score) || best_score < static_cast<float>(score_threshold)) {
1492 continue;
1493 }
1494
1495 const float box_a = value_at(0, anchor);
1496 const float box_b = value_at(1, anchor);
1497 const float box_c = value_at(2, anchor);
1498 const float box_d = value_at(3, anchor);
1499 if (!std::isfinite(box_a) || !std::isfinite(box_b) || !std::isfinite(box_c) || !std::isfinite(box_d)) {
1500 invalid_result = true;
1501 continue;
1502 }
1503 const cv::Rect2f model_box = yolo26 ? cv::Rect2f(box_a, box_b, box_c - box_a, box_d - box_b)
1504 : cv::Rect2f(box_a - box_c * 0.5f, box_b - box_d * 0.5f, box_c, box_d);
1505 cv::Rect2f original_box((model_box.x - letterbox.pad_x) / letterbox.scale,
1506 (model_box.y - letterbox.pad_y) / letterbox.scale, model_box.width / letterbox.scale,
1507 model_box.height / letterbox.scale);
1508 original_box = clipBox(original_box, image_width, image_height);
1509 if (original_box.area() <= 1.0f) {
1510 continue;
1511 }
1512
1513 Detection detection;
1514 detection.box = original_box;
1515 detection.class_index = best_class;
1516 detection.semantic_class_id = static_cast<int>(class_ids_[static_cast<std::size_t>(best_class)]);
1517 detection.score = best_score;
1518 detection.mask_coefficients.resize(static_cast<std::size_t>(mask_count));
1519 bool finite_coefficients = true;
1520 for (int64_t i = 0; i < mask_count; ++i) {
1521 const float coefficient = value_at(mask_channel_offset + i, anchor);
1522 finite_coefficients = finite_coefficients && std::isfinite(coefficient);
1523 detection.mask_coefficients[static_cast<std::size_t>(i)] = coefficient;
1524 }
1525 if (!finite_coefficients) {
1526 invalid_result = true;
1527 continue;
1528 }
1529 detections.push_back(std::move(detection));
1530 }
1531 };
1532
1533 switch (output_tensor_data_type_) {
1534 case TensorDataType::Fp32:
1535 decode_with_tensor(
1536 triton_interface.getOutputTensor<float>(yolo_detections_output_tensor_name_, shape[0], shape[1], shape[2]));
1537 break;
1538 case TensorDataType::Fp16:
1539 decode_with_tensor(
1540 triton_interface.getOutputTensor<Eigen::half>(yolo_detections_output_tensor_name_, shape[0], shape[1], shape[2]));
1541 break;
1542 }
1543 return detections;
1544}

1546std::vector<Detection> ImageInstanceSegmentation::decodeRfDetrDetections(triton_cpp::TritonInterface& triton_interface,
1547 int image_width,
1548 int image_height,
1549 double score_threshold,
1550 bool& invalid_result) {
1551 const auto boxes_shape = triton_interface.getOutputShape(rfdetr_boxes_output_tensor_name_);
1552 const auto logits_shape = triton_interface.getOutputShape(rfdetr_logits_output_tensor_name_);
1553 const int64_t num_classes = static_cast<int64_t>(class_names_.size());
1554 if (boxes_shape.size() != 3 || boxes_shape[0] != 1 || boxes_shape[2] != 4) {
1555 throw std::runtime_error("RF-DETR box output must have shape [1, queries, 4]");
1556 }
1557 const int64_t queries = boxes_shape[1];
1558 // RF-DETR exports may append an unused background logit.
1559 if (logits_shape.size() != 3 || logits_shape[0] != 1 || logits_shape[1] != queries ||
1560 (logits_shape[2] != num_classes && logits_shape[2] != num_classes + 1)) {
1561 throw std::runtime_error("RF-DETR logit output must have shape [1, queries, classes] or [1, queries, classes + 1]");
1562 }
1563
1564 std::vector<Detection> detections;
1565 visitTensorDataType(output_tensor_data_type_, [&](auto scalar) {
1566 using Scalar = decltype(scalar);
1567 const auto boxes = triton_interface.getOutputTensor<Scalar>(rfdetr_boxes_output_tensor_name_, 1, queries, 4);
1568 const auto logits = triton_interface.getOutputTensor<Scalar>(rfdetr_logits_output_tensor_name_, 1, queries, logits_shape[2]);
1569 for (int64_t query = 0; query < queries; ++query) {
1570 // Boxes are (cx, cy, w, h) normalized to the stretched input, i.e. to the original image.
1571 const float cx = static_cast<float>(boxes(0, query, 0)) * image_width;
1572 const float cy = static_cast<float>(boxes(0, query, 1)) * image_height;
1573 const float width = static_cast<float>(boxes(0, query, 2)) * image_width;
1574 const float height = static_cast<float>(boxes(0, query, 3)) * image_height;
1575 if (!std::isfinite(cx) || !std::isfinite(cy) || !std::isfinite(width) || !std::isfinite(height)) {
1576 invalid_result = true;
1577 continue;
1578 }
1579 const cv::Rect2f box = clipBox({cx - width * 0.5f, cy - height * 0.5f, width, height}, image_width, image_height);
1580 if (box.area() <= 1.0f) continue;
1581 for (int64_t cls = 0; cls < num_classes; ++cls) {
1582 const float logit = static_cast<float>(logits(0, query, cls));
1583 if (!std::isfinite(logit)) invalid_result = true;
1584 const float score = 1.0f / (1.0f + std::exp(-logit));
1585 if (!std::isfinite(score) || score < static_cast<float>(score_threshold)) continue;
1586 Detection detection;
1587 detection.box = box;
1588 detection.class_index = static_cast<int>(cls);

1599std::vector<Detection> ImageInstanceSegmentation::applyNms(std::vector<Detection> detections) const {
1600 std::vector<int> order(detections.size());
1601 std::iota(order.begin(), order.end(), 0);
1602 std::sort(order.begin(), order.end(), [&](int lhs, int rhs) {
1603 return detections[static_cast<std::size_t>(lhs)].score > detections[static_cast<std::size_t>(rhs)].score;
1604 });
1605
1606 std::vector<int> keep;
1607 std::vector<bool> suppressed(detections.size(), false);
1608 for (std::size_t i = 0; i < order.size(); ++i) {
1609 const int candidate = order[i];
1610 if (suppressed[static_cast<std::size_t>(candidate)]) {
1611 continue;
1612 }
1613 keep.push_back(candidate);
1614 for (std::size_t j = i + 1; j < order.size(); ++j) {
1615 const int other = order[j];
1616 if (suppressed[static_cast<std::size_t>(other)]) {
1617 continue;
1618 }
1619 if (detections[static_cast<std::size_t>(candidate)].class_index !=
1620 detections[static_cast<std::size_t>(other)].class_index) {
1621 continue;
1622 }
1623 if (intersectionOverUnion(detections[static_cast<std::size_t>(candidate)].box,
1624 detections[static_cast<std::size_t>(other)].box) > static_cast<float>(nms_iou_threshold_)) {
1625 suppressed[static_cast<std::size_t>(other)] = true;
1626 }
1627 }
1628 }
1629 std::vector<Detection> kept;
1630 kept.reserve(keep.size());
1631 for (int index : keep) {
1632 kept.push_back(std::move(detections[static_cast<std::size_t>(index)]));
1633 }
1634 return kept;
1635}

1651std::vector<InstanceMask> ImageInstanceSegmentation::reconstructYoloMasks(triton_cpp::TritonInterface& triton_interface,
1652 const std::vector<Detection>& detections,
1653 int image_width,
1654 int image_height,
1655 const LetterboxInfo& letterbox) {
1656 const auto runtime_shape = triton_interface.getOutputShape(yolo_prototypes_output_tensor_name_);
1657 if (!((runtime_shape.size() == 4 && runtime_shape[0] == 1) || runtime_shape.size() == 3)) {
1658 throw std::runtime_error("YOLO prototype output must have shape [1, masks, height, width]");
1659 }
1660 const std::array<int64_t, 4> shape =
1661 runtime_shape.size() == 4 ? std::array<int64_t, 4>{runtime_shape[0], runtime_shape[1], runtime_shape[2], runtime_shape[3]}
1662 : std::array<int64_t, 4>{1, runtime_shape[0], runtime_shape[1], runtime_shape[2]};
1663 const int64_t mask_count = shape[1];
1664 const int proto_height = static_cast<int>(shape[2]);
1665 const int proto_width = static_cast<int>(shape[3]);
1666 RCLCPP_DEBUG(get_logger(), "Prototype output shape: [%ld, %ld, %ld, %ld], mask_count=%ld", shape[0], shape[1], shape[2],
1667 shape[3], mask_count);
1668
1669 std::vector<InstanceMask> masks;
1670 masks.reserve(detections.size());
1671 const cv::Rect letterbox_roi(letterbox.pad_x, letterbox.pad_y, letterbox.resized_width, letterbox.resized_height);
1672 const cv::Rect proto_roi(
1673 static_cast<int>(std::round(letterbox_roi.x * proto_width / static_cast<double>(input_width_))),
1674 static_cast<int>(std::round(letterbox_roi.y * proto_height / static_cast<double>(input_height_))),
1675 static_cast<int>(std::round(letterbox_roi.width * proto_width / static_cast<double>(input_width_))),
1676 static_cast<int>(std::round(letterbox_roi.height * proto_height / static_cast<double>(input_height_))));
1677 const cv::Rect clipped_proto_roi = proto_roi & cv::Rect(0, 0, proto_width, proto_height);
1678
1679 const int proto_h = clipped_proto_roi.height;
1680 const int proto_w = clipped_proto_roi.width;
1681 if (proto_h <= 0 || proto_w <= 0) {
1682 throw std::runtime_error("Clipped prototype ROI is empty");
1683 }
1684 const int proto_pixels = proto_h * proto_w;
1685 cv::Mat prototypes_flat(static_cast<int>(mask_count), proto_pixels, CV_32F);
1686 auto fill_prototypes_flat = [&](auto prototypes) {
1687 using Scalar = std::remove_cv_t<std::remove_pointer_t<decltype(prototypes.data())>>;
1688 for (int64_t channel = 0; channel < mask_count; ++channel) {
1689 cv::Mat source(proto_height, proto_width, cvTypeFor<Scalar>(),
1690 const_cast<Scalar*>(prototypes.data()) + channel * proto_height * proto_width);
1691 cv::Mat destination = prototypes_flat.row(static_cast<int>(channel)).reshape(1, proto_h);
1692 source(clipped_proto_roi).convertTo(destination, CV_32F);
1693 }
1694 };
1695 switch (output_tensor_data_type_) {
1696 case TensorDataType::Fp32:
1697 fill_prototypes_flat(
1698 triton_interface.getOutputTensor<float>(yolo_prototypes_output_tensor_name_, shape[0], shape[1], shape[2], shape[3]));
1699 break;
1700 case TensorDataType::Fp16:
1701 fill_prototypes_flat(triton_interface.getOutputTensor<Eigen::half>(yolo_prototypes_output_tensor_name_, shape[0], shape[1],
1702 shape[2], shape[3]));
1703 break;
1704 }
1705
1706 cv::Mat coefficients(static_cast<int>(detections.size()), static_cast<int>(mask_count), CV_32F);
1707 for (std::size_t i = 0; i < detections.size(); ++i) {
1708 if (detections[i].mask_coefficients.size() != static_cast<std::size_t>(mask_count)) {
1709 throw std::runtime_error("Mask coefficient count does not match prototype channels");
1710 }
1711 std::memcpy(coefficients.ptr<float>(static_cast<int>(i)), detections[i].mask_coefficients.data(),
1712 static_cast<std::size_t>(mask_count) * sizeof(float));
1713 }
1714 cv::Mat all_logits;
1715 cv::gemm(coefficients, prototypes_flat, 1.0, cv::noArray(), 0.0, all_logits);
1716 for (std::size_t i = 0; i < detections.size(); ++i) {
1717 const cv::Mat logits = all_logits.row(static_cast<int>(i)).reshape(1, proto_h);
1718 masks.push_back(maskFromLogits(logits, detections[i].box, image_width, image_height, mask_threshold_));
1719 }
1720 return masks;
1721}

1751 }
1752 });

1801sensor_msgs::msg::Image ImageInstanceSegmentation::makeOverlay(const cv::Mat& bgr_image,
1802 const std::vector<Detection>& detections,
1803 const std::vector<InstanceMask>& masks,
1804 const std_msgs::msg::Header& header,
1805 OverlayColorMode color_mode) const {

1807
1808 // Alpha-blend each instance's mask color into the camera image, restricted to
1809 // that mask's pixels, so unsegmented background stays untouched instead of

1812 cv::Mat coverage(overlay.size(), CV_8UC1, cv::Scalar(0));

1817 coverage(masks[i].roi).setTo(255, masks[i].pixels);
1818 }
1819 cv::Mat blended;
1820 cv::addWeighted(overlay, 1.0 - overlay_alpha_, color_layer, overlay_alpha_, 0.0, blended);

1822
1823 if (overlay_draw_boxes_) {
1824 for (std::size_t i = 0; i < detections.size(); ++i) {
1825 const Detection& detection = detections[i];

1827 const cv::Scalar color = colorForInstance(color_index);
1828
1829 const cv::Rect box = boxRoi(detection.box, overlay.cols, overlay.rows);
1830 cv::rectangle(overlay, box, color, 2);
1831
1832 std::string label = detection.class_index >= 0 && static_cast<std::size_t>(detection.class_index) < class_names_.size()
1833 ? class_names_[static_cast<std::size_t>(detection.class_index)]
1834 : "?";
1835 if (tracker_config_.enabled && detection.track_id != 0) {
1836 label += " #" + std::to_string(detection.track_id);
1837 }
1838 char score_suffix[16];
1839 std::snprintf(score_suffix, sizeof(score_suffix), " %.2f", static_cast<double>(detection.score));
1840 label += score_suffix;
1841
1842 int baseline = 0;
1843 const cv::Size text_size = cv::getTextSize(label, cv::FONT_HERSHEY_SIMPLEX, 0.5, 1, &baseline);
1844 const int label_bottom = std::max(box.y, text_size.height + 4);
1845 cv::rectangle(overlay, cv::Point(box.x, label_bottom - text_size.height - 4),
1846 cv::Point(box.x + text_size.width + 4, label_bottom), color, cv::FILLED);
1847 cv::putText(overlay, label, cv::Point(box.x + 2, label_bottom - 2), cv::FONT_HERSHEY_SIMPLEX, 0.5, cv::Scalar(0, 0, 0), 1,
1848 cv::LINE_AA);
1849 }
1850 }
1851
1852 sensor_msgs::msg::Image msg;
1853 msg.header = header;
1854 msg.height = static_cast<std::uint32_t>(overlay.rows);
1855 msg.width = static_cast<std::uint32_t>(overlay.cols);
1856 msg.encoding = sensor_msgs::image_encodings::BGR8;
1857 msg.is_bigendian = false;
1858 msg.step = static_cast<sensor_msgs::msg::Image::_step_type>(overlay.cols * overlay.elemSize());
1859 const std::size_t size = static_cast<std::size_t>(msg.step) * msg.height;
1860 msg.data.resize(size);
1861 std::memcpy(msg.data.data(), overlay.data, size);
1862 return msg;
1863}
